Exported Module Code (math_utils.js)

export const PI = 3.14159265359;
export function getCircumference(radius) { return 2 * PI * radius; }
export function getArea(radius) { return PI * radius * radius; }
export function getSphereVolume(radius) { return (4/3) * PI * Math.pow(radius, 3); }
